Voici le code frontend à intégrer dans AG7 pour gérer les stories collaboratives par événement, sans backend.

Il comprend :

· De nouvelles modales HTML/CSS (création / rejoindre / affichage d’un événement).
· Des fonctions JavaScript pour créer, inviter, ajouter des stories et les visualiser (les appels API sont simulés, à remplacer par vos endpoints).
· L’intégration dans la section « Actus » (ou un onglet dédié « Événements »).

---

```javascript
function initCollabStories() {
  // Bouton "Événements" dans la barre stories
  const collabBtn = document.getElementById('collabEventsBtn');
  if (collabBtn) collabBtn.addEventListener('click', () => openMyEventsList());
  
  // Bouton créer événement dans la liste
  const createNewBtn = document.getElementById('createNewEventBtn');
  if (createNewBtn) createNewBtn.addEventListener('click', () => openCreateEventModal());
  
  // Fermeture des modales
  document.querySelectorAll('.modal-close').forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.target.closest('.modal').classList.add('hidden');
    });
  });
  
  // Gestion du formulaire création événement
  const createForm = document.getElementById('createEventForm');
  if (createForm) {
    createForm.addEventListener('submit', handleCreateEvent);
  }
  
  // Formulaire rejoindre événement (modale dédiée ou rapide)
  const joinForm = document.getElementById('joinEventForm');
  if (joinForm) joinForm.addEventListener('submit', handleJoinEvent);
  const quickJoinBtn = document.getElementById('quickJoinBtn');
  if (quickJoinBtn) quickJoinBtn.addEventListener('click', () => {
    const code = document.getElementById('quickInviteCode').value;
    if (code) handleJoinEventByCode(code);
  });
  
  // Aperçu cover
  const coverInput = document.getElementById('eventCover');
  if (coverInput) {
    coverInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (ev) => {
          const preview = document.getElementById('eventCoverPreview');
          preview.src = ev.target.result;
          preview.style.display = 'block';
        };
        reader.readAsDataURL(file);
      }
    });
  }
  
  // Contrôles du carrousel collaboratif
  const prevBtn = collabStoryModal?.querySelector('.story-prev');
  const nextBtn = collabStoryModal?.querySelector('.story-next');
  if (prevBtn) prevBtn.addEventListener('click', () => navigateCollabStory(-1));
  if (nextBtn) nextBtn.addEventListener('click', () => navigateCollabStory(1));
  
  const addBtn = collabStoryModal?.querySelector('#collabAddStoryBtn');
  if (addBtn) addBtn.addEventListener('click', () => openAddStoryToEvent(currentEventId));
  
  // Fermeture modale story
  const storyClose = collabStoryModal?.querySelector('.story-close');
  if (storyClose) storyClose.addEventListener('click', () => closeCollabStoryModal());
}
```

c) Gestion des événements (appels API simulés – à remplacer)

```javascript
// Remplacer les fetch par vos véritables endpoints
async function loadMyEvents() {
  try {
    // GET /api/events/my
    const response = await fetch('?action=getMyEvents');
    const data = await response.json();
    if (data.success) {
      myEvents = data.events;
      renderMyEventsList();
    }
  } catch (error) {
    console.error('Erreur chargement événements', error);
    // Fallback données de test
    myEvents = [
      { id: 1, name: 'Anniversaire Léa', role: 'owner', cover: null, start_date: '2025-06-10' },
      { id: 2, name: 'Soirée plage', role: 'contributor', cover: null, start_date: '2025-06-15' }
    ];
    renderMyEventsList();
  }
}

async function handleCreateEvent(e) {
  e.preventDefault();
  const formData = new FormData();
  formData.append('action', 'createEvent');
  formData.append('name', document.getElementById('eventName').value);
  formData.append('description', document.getElementById('eventDesc').value);
  formData.append('start_date', document.getElementById('eventStartDate').value);
  formData.append('end_date', document.getElementById('eventEndDate').value);
  formData.append('is_public', document.querySelector('input[name="eventPrivacy"]:checked').value === 'public' ? 1 : 0);
  const coverFile = document.getElementById('eventCover').files[0];
  if (coverFile) formData.append('cover_image', coverFile);
  
  try {
    const response = await fetch(window.location.href, { method: 'POST', body: formData });
    const result = await response.json();
    if (result.success) {
      showNotification('success', 'Événement créé', `Code d'invitation : ${result.invite_code}`);
      createEventModal.classList.add('hidden');
      await loadMyEvents();
    } else {
      showNotification('error', 'Erreur', result.message);
    }
  } catch (err) {
    console.error(err);
    showNotification('error', 'Erreur', 'Impossible de créer l’événement');
  }
}

async function handleJoinEvent(e) {
  e.preventDefault();
  const code = document.getElementById('inviteCode').value;
  await handleJoinEventByCode(code);
}

async function handleJoinEventByCode(code) {
  const formData = new FormData();
  formData.append('action', 'joinEvent');
  formData.append('invite_code', code);
  try {
    const response = await fetch(window.location.href, { method: 'POST', body: formData });
    const result = await response.json();
    if (result.success) {
      showNotification('success', 'Rejoint', `Vous participez maintenant à ${result.event_name}`);
      joinEventModal?.classList.add('hidden');
      collabEventsListModal?.classList.add('hidden');
      await loadMyEvents();
    } else {
      showNotification('error', 'Erreur', result.message);
    }
  } catch (err) {
    console.error(err);
    showNotification('error', 'Erreur', 'Impossible de rejoindre');
  }
}

function renderMyEventsList() {
  const container = document.getElementById('myEventsContainer');
  if (!container) return;
  if (myEvents.length === 0) {
    container.innerHTML = '<p>Aucun événement pour le moment.</p>';
    return;
  }
  container.innerHTML = myEvents.map(ev => `
    <div class="event-item" data-event-id="${ev.id}">
      <strong>${escapeHtml(ev.name)}</strong><br>
      <small>${ev.role === 'owner' ? 'Créateur' : 'Invité'} – ${new Date(ev.start_date).toLocaleDateString()}</small>
    </div>
  `).join('');
  container.querySelectorAll('.event-item').forEach(item => {
    item.addEventListener('click', () => {
      const eventId = parseInt(item.dataset.eventId);
      openEventStories(eventId);
    });
  });
}

async function openEventStories(eventId) {
  currentEventId = eventId;
  // Charger les stories de l'événement
  try {
    const response = await fetch(`?action=getEventStories&event_id=${eventId}`);
    const data = await response.json();
    if (data.success) {
      currentEventStories = data.stories; // tableau d'objets { id, user_name, user_photo, image_url, text, created_at }
      displayCollabStory(0);
      collabStoryModal.classList.remove('hidden');
      // Afficher/masquer bouton ajouter selon droit
      const addBtn = document.getElementById('collabAddStoryBtn');
      if (addBtn) addBtn.style.display = (data.can_contribute) ? 'block' : 'none';
      // Mettre à jour l'entête
      document.getElementById('collabEventTitle').innerText = data.event_name;
      document.getElementById('collabEventMeta').innerText = `${data.stories_count} stories · ${data.members_count} participants`;
    } else {
      showNotification('error', 'Erreur', data.message);
    }
  } catch (err) {
    console.error(err);
    showNotification('error', 'Erreur', 'Impossible de charger les stories');
  }
}

function displayCollabStory(index) {
  if (currentEventStories.length === 0) {
    document.getElementById('collabStoryImage').src = '';
    document.getElementById('collabStoryCaption').innerText = 'Aucune story pour le moment. Soyez le premier à ajouter !';
    document.getElementById('collabStoryAuthor').innerText = '';
    return;
  }
  const story = currentEventStories[index];
  currentEventIndex = index;
  const imgElement = document.getElementById('collabStoryImage');
  imgElement.src = story.image_url ? `story/${story.image_url}` : '';
  document.getElementById('collabStoryCaption').innerText = story.text || '';
  document.getElementById('collabStoryAuthor').innerHTML = `<i class="fas fa-user-circle"></i> ${escapeHtml(story.user_name)}`;
  
  // Démarrer le timer de progression (5 secondes)
  if (eventStoryTimer) clearInterval(eventStoryTimer);
  const fill = document.querySelector('.story-progress-fill');
  fill.style.width = '0%';
  let startTime = Date.now();
  eventStoryTimer = setInterval(() => {
    const elapsed = (Date.now() - startTime) / 1000;
    const percent = (elapsed / 5) * 100;
    fill.style.width = Math.min(percent, 100) + '%';
    if (elapsed >= 5) {
      clearInterval(eventStoryTimer);
      navigateCollabStory(1);
    }
  }, 50);
}

function navigateCollabStory(direction) {
  if (currentEventStories.length === 0) return;
  let newIndex = currentEventIndex + direction;
  if (newIndex < 0) newIndex = 0;
  if (newIndex >= currentEventStories.length) {
    closeCollabStoryModal();
    return;
  }
  displayCollabStory(newIndex);
}

function closeCollabStoryModal() {
  if (eventStoryTimer) clearInterval(eventStoryTimer);
  collabStoryModal.classList.add('hidden');
}

async function openAddStoryToEvent(eventId) {
  // Utiliser la modale existante de création de story (createStoryModal) mais en mode événement
  // Pour simplifier, on affiche la modale de story normale et on ajoute un champ caché event_id
  const storyModal = document.getElementById('createStoryModal');
  if (!storyModal) return;
  // Ajouter un champ hidden dans le formulaire
  let hiddenInput = document.getElementById('storyEventId');
  if (!hiddenInput) {
    hiddenInput = document.createElement('input');
    hiddenInput.type = 'hidden';
    hiddenInput.id = 'storyEventId';
    storyModal.querySelector('.create-story-form').appendChild(hiddenInput);
  }
  hiddenInput.value = eventId;
  // Modifier le bouton de publication pour envoyer vers l'événement
  const originalPublish = document.getElementById('publishStoryBtn');
  const newPublish = originalPublish.cloneNode(true);
  originalPublish.parentNode.replaceChild(newPublish, originalPublish);
  newPublish.addEventListener('click', async () => {
    const text = document.getElementById('storyText').value;
    const imageFile = selectedStoryFile;
    const formData = new FormData();
    formData.append('action', 'addEventStory');
    formData.append('event_id', eventId);
    if (text) formData.append('story_text', text);
    if (imageFile) formData.append('story_image', imageFile);
    try {
      const response = await fetch(window.location.href, { method: 'POST', body: formData });
      const result = await response.json();
      if (result.success) {
        showNotification('success', 'Story ajoutée', 'Votre contribution est en ligne');
        storyModal.classList.add('hidden');
        await openEventStories(eventId); // recharger
      } else {
        showNotification('error', 'Erreur', result.message);
      }
    } catch (err) {
      console.error(err);
      showNotification('error', 'Erreur', 'Impossible d’ajouter la story');
    }
  });
  storyModal.classList.remove('hidden');
}

function openMyEventsList() {
  loadMyEvents();
  collabEventsListModal.classList.remove('hidden');
}

function openCreateEventModal() {
  createEventModal.classList.remove('hidden');
}
```
